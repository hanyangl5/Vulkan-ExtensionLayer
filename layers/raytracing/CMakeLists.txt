cmake_minimum_required(VERSION 3.22.1)
project(VELRayTracingFallback LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The experimental RT fallback currently requires a 64-bit build")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
find_package(Vulkan REQUIRED)
find_package(SPIRV-Tools CONFIG REQUIRED)
find_package(SPIRV-Tools-opt CONFIG REQUIRED)
find_package(SPIRV-Tools-link CONFIG REQUIRED)
if(NOT TARGET SPIRV-Tools AND TARGET SPIRV-Tools-static)
    add_library(SPIRV-Tools ALIAS SPIRV-Tools-static)
endif()
find_package(glslang CONFIG REQUIRED)
find_package(spirv_cross_core CONFIG REQUIRED)
find_package(spirv_cross_glsl CONFIG REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/continuation_runtime.glsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/continuation_runtime.glsl" CONTINUATION_RUNTIME_SOURCE)
configure_file(continuation_runtime.h.in continuation_runtime.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/bvh.glsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/bvh.glsl" BVH_SHADER_SOURCE)
configure_file(bvh_shader.h.in bvh_shader.h @ONLY)
add_library(vel_rt_compiler STATIC compiler.cpp continuations.cpp bvh.cpp)
target_link_libraries(vel_rt_compiler PUBLIC spirv-cross-glsl spirv-cross-core glslang::glslang glslang::SPIRV
    glslang::glslang-default-resource-limits SPIRV-Tools-opt SPIRV-Tools-link SPIRV-Tools-static)
target_include_directories(vel_rt_compiler PUBLIC . ${CMAKE_CURRENT_BINARY_DIR} ${Vulkan_INCLUDE_DIRS})
add_library(VkLayer_hanyang_ray_tracing_fallback SHARED layer.cpp)
target_link_libraries(VkLayer_hanyang_ray_tracing_fallback PRIVATE vel_rt_compiler)
target_include_directories(VkLayer_hanyang_ray_tracing_fallback PRIVATE ${Vulkan_INCLUDE_DIRS})
if(UNIX AND NOT APPLE)
    target_link_options(VkLayer_hanyang_ray_tracing_fallback PRIVATE -Wl,-Bsymbolic,--exclude-libs,ALL)
endif()
configure_file(VkLayer_hanyang_ray_tracing_fallback.json.in VkLayer_hanyang_ray_tracing_fallback.json @ONLY)
enable_testing()
add_executable(vel_rt_tests tests.cpp)
target_link_libraries(vel_rt_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_dependencies(vel_rt_tests VkLayer_hanyang_ray_tracing_fallback)
foreach(variant u16 u32 nonindexed empty runtime builder metadata tlas)
    add_test(NAME ray_tracing_fallback_${variant} COMMAND vel_rt_tests ${variant})
    set_tests_properties(ray_tracing_fallback_${variant} PROPERTIES ENVIRONMENT "VK_LAYER_PATH=${CMAKE_CURRENT_BINARY_DIR}")
endforeach()

add_executable(vel_rt_continuation_tests continuation_tests.cpp)
target_link_libraries(vel_rt_continuation_tests PRIVATE vel_rt_compiler)
add_test(NAME ray_tracing_fallback_continuation_analysis COMMAND vel_rt_continuation_tests)

find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME ray_tracing_fallback_pinned_sources COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/upstream/verify.py")
