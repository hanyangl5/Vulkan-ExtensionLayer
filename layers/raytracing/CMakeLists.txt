cmake_minimum_required(VERSION 3.22.1)
project(VELRayTracingFallback LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The experimental RT fallback currently requires a 64-bit build")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
find_package(Vulkan REQUIRED)
# Share the registry-matched pNext size metadata with the RQ host adapter.
include("${CMAKE_CURRENT_SOURCE_DIR}/../../vkrt_layer/layer/pnext_sizes.cmake")
find_package(SPIRV-Tools CONFIG REQUIRED)
find_package(SPIRV-Tools-opt CONFIG REQUIRED)
find_package(SPIRV-Tools-link CONFIG REQUIRED)
if(NOT TARGET SPIRV-Tools AND TARGET SPIRV-Tools-static)
    add_library(SPIRV-Tools ALIAS SPIRV-Tools-static)
endif()
find_package(glslang CONFIG REQUIRED)
find_package(spirv_cross_core CONFIG REQUIRED)
find_package(spirv_cross_glsl CONFIG REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/continuation_runtime.glsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/continuation_runtime.glsl" CONTINUATION_RUNTIME_SOURCE)
configure_file(continuation_runtime.h.in continuation_runtime.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/bvh.glsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/bvh.glsl" BVH_SHADER_SOURCE)
configure_file(bvh_shader.h.in bvh_shader.h @ONLY)
include("${CMAKE_CURRENT_SOURCE_DIR}/typed_runtime_sources.cmake")
add_library(vel_rt_compiler STATIC compiler.cpp glsl_compiler.cpp continuations.cpp frame_emitter.cpp state_compiler.cpp typed_pipeline.cpp typed_runtime.cpp bvh.cpp)
target_link_libraries(vel_rt_compiler PUBLIC spirv-cross-glsl spirv-cross-core glslang::glslang glslang::SPIRV
    glslang::glslang-default-resource-limits SPIRV-Tools-opt SPIRV-Tools-link SPIRV-Tools-static)
target_include_directories(vel_rt_compiler PUBLIC . ${CMAKE_CURRENT_BINARY_DIR} ${Vulkan_INCLUDE_DIRS})
add_library(VkLayer_hanyang_ray_tracing_fallback SHARED layer.cpp)
target_link_libraries(VkLayer_hanyang_ray_tracing_fallback PRIVATE vel_rt_compiler)
target_include_directories(VkLayer_hanyang_ray_tracing_fallback PRIVATE ${Vulkan_INCLUDE_DIRS})
if(UNIX AND NOT APPLE)
    target_link_options(VkLayer_hanyang_ray_tracing_fallback PRIVATE -Wl,-Bsymbolic,--exclude-libs,ALL)
endif()
# Resolve the actual target path so a layer enabled from another working
# directory loads correctly, including Windows DLL-search semantics.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/VkLayer_hanyang_ray_tracing_fallback.json.in")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VkLayer_hanyang_ray_tracing_fallback.json.in" _vel_rt_manifest_template)
set(VEL_RT_LIBRARY_PATH "$<TARGET_FILE:VkLayer_hanyang_ray_tracing_fallback>")
string(CONFIGURE "${_vel_rt_manifest_template}" _vel_rt_manifest @ONLY)
get_property(_vel_rt_multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(_vel_rt_multi_config)
    set(_vel_rt_manifest_dir "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
else()
    set(_vel_rt_manifest_dir "${CMAKE_CURRENT_BINARY_DIR}")
endif()
file(GENERATE OUTPUT "${_vel_rt_manifest_dir}/VkLayer_hanyang_ray_tracing_fallback.json" CONTENT "${_vel_rt_manifest}")
enable_testing()
add_executable(vel_rt_tests tests.cpp)
target_link_libraries(vel_rt_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_dependencies(vel_rt_tests VkLayer_hanyang_ray_tracing_fallback)
foreach(variant u16 u32 nonindexed empty runtime builder metadata tlas device_features)
    add_test(NAME ray_tracing_fallback_${variant} COMMAND vel_rt_tests ${variant})
    set_tests_properties(ray_tracing_fallback_${variant} PROPERTIES ENVIRONMENT "VK_LAYER_PATH=${_vel_rt_manifest_dir}")
endforeach()
set_tests_properties(ray_tracing_fallback_device_features PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../..")

add_executable(vel_rt_continuation_tests continuation_tests.cpp)
target_link_libraries(vel_rt_continuation_tests PRIVATE vel_rt_compiler)
add_test(NAME ray_tracing_fallback_continuation_analysis COMMAND vel_rt_continuation_tests)

add_executable(vel_rt_frame_emitter_tests frame_emitter_tests.cpp)
target_link_libraries(vel_rt_frame_emitter_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_test(NAME ray_tracing_fallback_typed_frame_execution COMMAND vel_rt_frame_emitter_tests)
set_tests_properties(ray_tracing_fallback_typed_frame_execution PROPERTIES SKIP_RETURN_CODE 77)

add_executable(vel_rt_state_compiler_tests state_compiler_tests.cpp)
target_link_libraries(vel_rt_state_compiler_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_test(NAME ray_tracing_fallback_typed_source_cfg COMMAND vel_rt_state_compiler_tests)
set_tests_properties(ray_tracing_fallback_typed_source_cfg PROPERTIES SKIP_RETURN_CODE 77)

add_executable(vel_rt_request_tests rt_request_tests.cpp)
target_link_libraries(vel_rt_request_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_test(NAME ray_tracing_fallback_typed_rt_requests COMMAND vel_rt_request_tests)
set_tests_properties(ray_tracing_fallback_typed_rt_requests PROPERTIES SKIP_RETURN_CODE 77)

add_executable(vel_rt_typed_pipeline_tests typed_pipeline_tests.cpp)
target_link_libraries(vel_rt_typed_pipeline_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_test(NAME ray_tracing_fallback_typed_pipeline COMMAND vel_rt_typed_pipeline_tests)
set_tests_properties(ray_tracing_fallback_typed_pipeline PROPERTIES SKIP_RETURN_CODE 77)

add_executable(vel_rt_typed_bda_tests typed_bda_tests.cpp)
target_link_libraries(vel_rt_typed_bda_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_test(NAME ray_tracing_fallback_typed_bda COMMAND vel_rt_typed_bda_tests)
set_tests_properties(ray_tracing_fallback_typed_bda PROPERTIES SKIP_RETURN_CODE 77)
add_executable(vel_rt_typed_runtime_tests typed_runtime_tests.cpp)
target_link_libraries(vel_rt_typed_runtime_tests PRIVATE vel_rt_compiler Vulkan::Vulkan)
add_test(NAME ray_tracing_fallback_typed_runtime COMMAND vel_rt_typed_runtime_tests)
set_tests_properties(ray_tracing_fallback_typed_runtime PROPERTIES SKIP_RETURN_CODE 77)

add_test(NAME ray_tracing_fallback_pinned_sources COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/upstream/verify.py")
